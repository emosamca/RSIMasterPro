//@version=5
strategy("RSI Master Pro - KUSURSUZ FINAL V2", overlay=true, initial_capital=1000, default_qty_type=strategy.cash, currency=currency.USD)

// ==========================================
// --- TARİH ARALIĞI ---
// ==========================================
startDate = input.time(timestamp("2025-01-01 00:00"), "Başlangıç Tarihi")
endDate   = input.time(timestamp("2025-12-31 23:59"), "Bitiş Tarihi")

dateFilter = time >= startDate and time <= endDate

// ==========================================
// --- GİRDİLER ---
// ==========================================
len_fast    = input.int(9, "Hızlı RSI")
len_slow    = input.int(18, "Yavaş RSI")
ema_len     = input.int(80, "Ana Trend EMA")
adx_min     = input.int(20, "Minimum ADX")

// Risk ve Kar Ayarları
tp_mult     = input.float(6.0, "Kar Al (ATR)", step=0.1)
sl_mult     = input.float(0.8, "Zarar Durdur (ATR)", step=0.1)
risk_per_trade = input.float(3.0, "İşlem Başı Risk %")

// ==========================================
// --- HESAPLAMALAR ---
// ==========================================
src = close
ema200 = ta.ema(src, ema_len)
atr = ta.atr(14)
rsi_main = ta.rsi(src, 14)

// Divergence ve İvme
div = ta.rsi(src, len_fast) - ta.rsi(src, len_slow)
div_accel = div - div[1]

// Gelişmiş Filtreler
[diplus, diminus, adx] = ta.dmi(14, 14)
is_trending = adx > adx_min

vol_sma = ta.sma(volume, 20)
vol_force = volume > vol_sma * 1.2

// Momentum Onayı
strong_long  = rsi_main > 55 and div_accel > 0.1
strong_short = rsi_main < 45 and div_accel < -0.1

// ==========================================
// --- DİNAMİK RİSK YÖNETİMİ ---
// ==========================================
float risk_amount = strategy.equity * (risk_per_trade / 100)
float stop_dist = atr * sl_mult
float pos_size = stop_dist > 0 ? risk_amount / stop_dist : 0

// ==========================================
// --- GİRİŞ KOŞULLARI ---
// ==========================================
longCondition  = ta.crossover(div, 0) and src > ema200 and is_trending and vol_force and strong_long
shortCondition = ta.crossunder(div, 0) and src < ema200 and is_trending and vol_force and strong_short

// ==========================================
// --- STRATEJİ UYGULAMA ---
// ==========================================
if (longCondition and dateFilter)
    sl = src - (atr * sl_mult)
    tp = src + (atr * tp_mult)

    strategy.entry("Long", strategy.long, qty=pos_size, comment="Strong Bull")

    strategy.exit(
         "Exit L",
         "Long",
         stop=sl,
         limit=tp,
         trail_points=atr * 6,
         trail_offset=atr * 2
     )

if (shortCondition and dateFilter)
    sl = src + (atr * sl_mult)
    tp = src - (atr * tp_mult)

    strategy.entry("Short", strategy.short, qty=pos_size, comment="Strong Bear")

    strategy.exit(
         "Exit S",
         "Short",
         stop=sl,
         limit=tp,
         trail_points=atr * 6,
         trail_offset=atr * 2
     )

// ==========================================
// --- GÖRSELLEŞTİRME ---
// ==========================================
plot(ema200, color=#FFD700, linewidth=2, title="Safe Zone")

plotshape(longCondition and dateFilter,     style=shape.triangleup,     location=location.belowbar,     color=#00FF00,     size=size.small)

plotshape(     shortCondition and dateFilter,     style=shape.triangledown,     location=location.abovebar,     color=#FF0000,     size=size.small)

// Hacimsiz ve zayıf bölgeleri renklendirme
bgcolor(not vol_force or not is_trending ? color.new(#808080, 95) : na)
